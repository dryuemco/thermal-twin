\begin{figure}[htbp]
  \centering
  \makebox[\linewidth][c]{\includegraphics{figures/fig2_schematic.pdf}}
  \caption{\textbf{Processing chain and evaluation design.} Five source products are combined on
  the $\sim$510\,m analysis grid, and only regions that pass the burned-landcover gate are
  modelled; Kozan 2023 fails it, because 98\% of its burned area is cropland. The baseline and
  thermal predictor sets are evaluated within regions by spatially blocked cross-validation and
  across regions in 20 directions, without and with label-free adaptation, and the scores are
  computed on the whole region, on scar frames and on distance collars. The small grey numbers give
  the Methods section. Leakage control, seed 42 and the spatial-block bootstrap (Section 2.13) apply
  throughout.}
  \label{fig:pipeline}

% ---------------------------------------------------------------- Figure 3 --
\begin{figure}[htbp]
  \centering
  \makebox[\linewidth][c]{\includegraphics{figures/fig3_within_robustness.pdf}}
  \caption{\textbf{The within-region thermal gain in all five regions, by block size.}
  Natural-vegetation population; random forest; spatially blocked five-fold cross-validation. Each
  panel shows one region: the area under the receiver operating characteristic curve (ROC-AUC) of
  the baseline and thermal sets with blocks of 2, 10 and 20 cells ($\approx$1, 5 and 10\,km). The shaded band is the thermal gain. Skill falls as the
  blocks become larger, and the thermal set stays above the baseline at every block size. The 95\%
  spatial-block bootstrap interval of the gain (1000 replicates, seed 42) excludes zero in all
  regions at 2 and 10 cells. At 20 cells only 6 to 33 blocks contain burned cells, so these values
  are indicative (Table~\ref{tab:2}, note).}
  \label{fig:within-robustness}
\end{figure}

\begin{figure}[htbp]
  \centering
  \makebox[\linewidth][c]{\includegraphics{figures/fig7_feature_drop.pdf}}
  \caption{\textbf{Removing the reversing features costs local skill and gives no measurable
  transfer gain.} Four feature sets: all ten predictors, without the land surface temperature (LST)
  anomaly, without elevation, and without both. \textbf{(a)} Mean within-region ROC-AUC over five
  regions at 1\,km blocking. \textbf{(b)} Mean transfer ROC-AUC over 20 directions. Both axes span
  0.12 ROC-AUC, and the vertical lines mark the full set. Within-region skill falls
  (0.896 $\rightarrow$ 0.820), a cost of 0.076 that is supported in every region, while mean
  transfer rises (0.527 $\rightarrow$ 0.541). The horizontal line in (b) is the full-set level plus
  the pair-$t$ interval of this change over ten region pairs, [$-$0.028, $+$0.056]; it crosses the
  vertical line, so zero change is included (the primary pair-cluster bootstrap gives
  [$-$0.018, $+$0.049]).}
  \label{fig:feature-drop}
\end{figure}
